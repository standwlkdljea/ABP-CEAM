USE vetcuidado_db;

-- Asegúrate de que la tabla usuarios tenga contraseña
CREATE TABLE usuarios (
  id int NOT NULL AUTO_INCREMENT,
  nombre_completo varchar(100) NOT NULL,
  dni varchar(20) NOT NULL UNIQUE,
  email varchar(100) NOT NULL UNIQUE,
  password_hash varchar(255) NOT NULL, -- Para el login
  telefono varchar(20) DEFAULT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Asegúrate de que la tabla doctores tenga email y contraseña
CREATE TABLE doctores (
  id int NOT NULL AUTO_INCREMENT,
  nombre_doctor varchar(100) NOT NULL,
  especialidad varchar(50) NOT NULL,
  email varchar(100) NOT NULL UNIQUE,      -- Para el login del staff
  password_hash varchar(255) NOT NULL, -- Para el login del staff
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- Tabla de Mascotas (Ahora vinculada al usuario/dueño mediante id_usuario)
CREATE TABLE mascotas (
  id int NOT NULL AUTO_INCREMENT,
  nombre_mascota varchar(50) COLLATE utf8mb4_unicode_ci NOT NULL,
  tipo_mascota varchar(30) COLLATE utf8mb4_unicode_ci NOT NULL,
  edad int NOT NULL,
  id_usuario int NOT NULL, -- Relación con el dueño
  PRIMARY KEY (id),
  KEY idx_mascotas_usuario (id_usuario),
  CONSTRAINT fk_mascotas_usuarios FOREIGN KEY (id_usuario) REFERENCES usuarios (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- =========================================================================

-- Tabla de Citas (Limpia de datos redundantes del cliente o mascota)
CREATE TABLE citas (
  id int NOT NULL AUTO_INCREMENT,
  fecha_cita datetime NOT NULL,
  motivo text COLLATE utf8mb4_unicode_ci NOT NULL,
  id_mascota int NOT NULL,
  id_doctor int NOT NULL,
  PRIMARY KEY (id),
  KEY idx_citas_doctor (id_doctor),
  KEY idx_citas_mascota (id_mascota),
  CONSTRAINT fk_citas_doctores FOREIGN KEY (id_doctor) REFERENCES doctores (id) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT fk_citas_mascotas FOREIGN KEY (id_mascota) REFERENCES mascotas (id) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
